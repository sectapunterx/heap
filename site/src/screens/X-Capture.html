if err != nil {
return fmt.Errorf("charge: %w", err)
}
if resp.StatusCode == 429 {
wait := parseRetryAfter(resp.Header.Get("Retry-After"))
return &RateLimited{Wait: wait}
}
defer resp.Body.Close()
return decode(resp.Body, &out)
}
